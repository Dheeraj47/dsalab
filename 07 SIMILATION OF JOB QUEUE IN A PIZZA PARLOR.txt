#include <stdio.h>
#define MAX 5

int queue[MAX];
int front = -1, rear = -1;

// Function to add an order (Enqueue)
void add()
{
    int order;

    // Check whether queue is full
    if (rear == MAX - 1)
        printf("Queue is Full!\n");
    else
    {
        printf("Enter order number: ");
        scanf("%d", &order);

        // If first order, set front to 0
        if (front == -1)
            front = 0;

        // Insert order and increase rear
        queue[++rear] = order;

        printf("Order Added!\n");
    }
}

// Function to serve an order (Dequeue)
void serve()
{
    // Check whether queue is empty
    if (front == -1 || front > rear)
        printf("No Orders!\n");
    else
    {
        // Serve the first order and move front forward
        printf("Serving Order: %d\n", queue[front++]);

        // Reset queue when all orders are served
        if (front > rear)
            front = rear = -1;
    }
}

// Function to display pending orders
void display()
{
    int i;

    if (front == -1)
        printf("No Orders!\n");
    else
    {
        // Print all orders from front to rear
        for (i = front; i <= rear; i++)
            printf("%d ", queue[i]);

        printf("\n");
    }
}

int main()
{
    int ch;

    // Display menu repeatedly
    do
    {
        printf("\n1.Add Order\n2.Serve Order\n3.Display\n4.Exit\n");
        printf("Enter choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
            case 1: add(); break;
            case 2: serve(); break;
            case 3: display(); break;
            case 4: printf("Exit\n"); break;
            default: printf("Invalid Choice!\n");
        }

    } while (ch != 4);

    return 0;
}
