#include <stdio.h>

int main()
{
    // Declare adjacency matrix and variables
    int a[5][5], n, i, j;

    // Take number of airports
    printf("Enter number of airports: ");
    scanf("%d", &n);

    // Take flight connections as input
    printf("Enter matrix (0 or 1):\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < n; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    // Display adjacency matrix
    printf("Flight Connections:\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < n; j++)
        {
            printf("%d ", a[i][j]);
        }
        printf("\n");
    }

    // Display direct flight connections
    printf("Direct Flights:\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < n; j++)
        {
            if(a[i][j] == 1)
            {
                printf("%d -> %d\n", i+1, j+1);
            }
        }
    }

    return 0;
}
